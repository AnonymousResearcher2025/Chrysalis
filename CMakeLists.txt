cmake_minimum_required(VERSION 3.20)
project(chrysalis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
list(APPEND CMAKE_PREFIX_PATH "${PYBIND_DIR}")
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_core cpp/core.cpp)
install(TARGETS _core LIBRARY DESTINATION chrysalis RUNTIME DESTINATION chrysalis)
